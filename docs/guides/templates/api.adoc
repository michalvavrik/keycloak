<#function escapeAsciidoc text>
  <#return text?replace("{", "\\{")>
</#function>
<#macro endpoint op>
<#assign escapedPath = escapeAsciidoc(op.path)>
===== ${op.httpMethod} `${escapedPath}`

${(op.summary)!}

<#if (op.description)?has_content>
${op.description}

</#if>
====== HTTP Request

[source,http]
----
${op.httpMethod} ${op.path} HTTP/1.1
<#if (op.requestBody.contentType)??>
Content-Type: ${op.requestBody.contentType}
</#if>
----

<#assign cliEx = (ctx.adminApiV2.getCliExample(op.operationId))!>
<#if cliEx?has_content>
====== CLI

[source,bash]
----
${cliEx}
----

</#if>
<#if (op.javaExample)?has_content>
====== Java Client

[source,java]
----
${op.javaExample}
----

</#if>
<#assign jsEx = (ctx.adminApiV2.getJsExample(op.operationId))!>
<#if jsEx?has_content>
====== JavaScript Client

[source,javascript]
----
${jsEx}
----

</#if>
<#if (op.parameters)?has_content>
====== Parameters

<#list op.parameters as param>
* `${param.name}` (${(param.in)!}) — ${(param.description)!}
</#list>

</#if>
<#if (op.responses)??>
====== Responses

<#list op.responses?keys as code>
* `${code}` — ${(op.responses[code].description)!}
</#list>

</#if>
</#macro>
